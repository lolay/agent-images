# Every Linux image's session tools and build prerequisites, from Ubuntu's
# archive. No versions on purpose: each build installs the current release, and
# a version gets pinned only when a release breaks something (see "Track the
# latest" in specs/design.md). # starts a comment.

# The rest of the session tools are in core/Brewfile, shared with macOS. These
# are the ones the system side needs before Homebrew is on PATH: Homebrew's
# installer (git, curl), the hooks and image scripts (jq), and archives.
ca-certificates
curl
git
jq
python3
unzip
zip

# Man page lint (nowline's lint-man, triage's release).
mandoc

# GNU time (`/usr/bin/time -v` for peak memory) and bc, which build and test
# scripts expect from a Linux box.
bc
time

# Homebrew's prerequisites on Ubuntu (git and curl are above). The Brewfiles'
# formulae are prebuilt, but Homebrew refuses to install a formula without a
# system compiler ("No developer tools installed"), and sessions need one anyway
# (Rust links with cc).
build-essential
file
procps
