# Binary checks (toolset-checks.sh): label: command, which must exit 0; its first
# line of output is the reported version. None needs a GPU, so they run on a
# hosted runner too; smoke.sh runs the GPU itself.
nvcc: nvcc --version | grep -m1 release
cuda-probe: command -v agent-cuda-probe
driver-libs: command -v agent-gpu-libs
eglinfo: command -v eglinfo
vulkaninfo: command -v vulkaninfo
virtualgl: vglrun -h 2>&1 | grep -m1 -i virtualgl || ls /opt/VirtualGL/bin/vglrun
