# Binary checks (toolset-checks.sh): label: command, which must exit 0; its first
# line of output is the reported version. The smoke VM's emulator boot
# (agent-gce-smoke) is this toolset's smoke test.
adb: adb version | head -n 1
gradle: gradle --version | grep -m1 '^Gradle'
# JDK 17 beside the jvm toolset's 21 (packages.txt).
java 17: /usr/lib/jvm/java-17-openjdk-*/bin/java -version
