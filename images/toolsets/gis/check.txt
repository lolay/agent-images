# Binary checks (toolset-checks.sh): label: command, which must exit 0; its first
# line of output is the reported version. smoke.sh shows GDAL and the Python
# environment work.
gdal: gdalinfo --version
pdal: pdal --version | grep -m1 '^pdal'
qgis: qgis_process --version 2>/dev/null | grep -m1 QGIS
python3: python3 --version
uv: uv --version
ruff: ruff --version
pre-commit: pre-commit --version
tippecanoe: tippecanoe --version 2>&1 | head -n 1
untwine: command -v untwine
osmium: osmium --version | head -n 1
dropped plugins: wc -l < /opt/gis/dropped-plugins.txt | sed 's/$/ dropped (see \/opt\/gis\/dropped-plugins.txt)/'
