# Smoke checks (images/cloud/entrypoint.sh): label: command, which must exit 0;
# its first line of output is the reported version.
gdal: gdalinfo --version
pdal: pdal --version | grep -m1 '^pdal'
qgis: qgis_process --version 2>/dev/null | grep -m1 QGIS
python3: python3 --version
# The imports must also print no errors: PDAL reports a plugin it can't load and
# carries on.
bindings: out="$(python3 -c 'import osgeo.gdal, rasterio, geopandas, pdal, qgis.core, pystac_client, duckdb' 2>&1)" && [ -z "$out" ] && echo "osgeo, rasterio, geopandas, pdal, qgis.core, pystac_client, duckdb" || { echo "$out"; false; }
one gdal: python3 -c 'import osgeo.gdal, rasterio; a, b = osgeo.gdal.__version__, rasterio.__gdal_version__; assert a == b, (a, b); print("GDAL", a)'
uv: uv --version
