# Packages installed into every Linux image, from Ubuntu's archive. No versions on
# purpose: each build installs the current release, and a version gets pinned
# only when a release breaks something (see "Track the latest" in
# specs/design.md). install-packages.sh reads this file; # starts a comment.

ca-certificates

# Man page lint (nowline's lint-man, triage's release). The language toolchains
# come from the Brewfile, which has the current releases; Ubuntu's are years old.
mandoc

# GNU time (`/usr/bin/time -v` for peak memory) and bc, which build and test
# scripts expect from a Linux box.
bc
time

# Homebrew's prerequisites on Ubuntu (git and curl are above). The Brewfile's
# formulae are prebuilt, but Homebrew refuses to install a formula without a
# system compiler ("No developer tools installed"), and sessions need one anyway
# (Rust links with cc).
build-essential
file
procps

# KVM checks (kvm-ok) for make doctor and agent-emulator status.
cpu-checker

# Android builds. JDK 21 is the version Android Studio bundles.
